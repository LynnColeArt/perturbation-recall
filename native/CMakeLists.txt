cmake_minimum_required(VERSION 3.18)
project(perturbation_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT LLAMA_SOURCE_DIR)
    message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned llama.cpp checkout")
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SOURCE_DIR} llama-build)
add_executable(perturbation-worker worker.cpp)
target_include_directories(perturbation-worker PRIVATE ${LLAMA_SOURCE_DIR}/vendor)
target_link_libraries(perturbation-worker PRIVATE llama ggml)
